Critical Rules

NONever call loadData() after field saves — Race condition, dates snap back. Update local state with setSelectedTask + setDetyrat instead.
NODate inputs save on onChange, not onBlur — onBlur races with loadData() re-fetch.

NONo template literals in JSX <option> text — Turbopack build fails.
NOconfirm() is blocked on Vercel — Use inline state-based confirmation UI instead.

Navigation Pattern — Mobile

💡
Recommended: Bottom tab bar for mobile (like Fieldwire iOS). Tabs: Dashboard · Tasks · Plans · Photos · More. The current left sidebar should be hidden on mobile and replaced with a bottom nav + hamburger for secondary pages. This matches what field workers expect from a mobile app.

Touch UX Rules

  • Minimum tap target: 44×44px (Apple HIG). All buttons, links, task rows.
  • No hover-only interactions — all hover states must have a tap equivalent.
  • Bottom sheets instead of side panels for task/document detail on mobile.
  • Sticky action buttons at bottom of screen (not top) — reachable with thumb.
  • Font sizes minimum 16px for body text on mobile (prevents iOS auto-zoom on inputs).
  • Input fields: font-size 16px minimum to prevent iOS zoom-on-focus.
  • Drag-and-drop upload: fall back to <input type="file" capture="environment"> for camera access.
  • Gantt view: hide on mobile (too dense). Show "Gantt available on desktop" message.

-- S11 RLS policies
DROP POLICY IF EXISTS "projektet_member_read" ON projektet;
CREATE POLICY "projektet_member_read" ON projektet FOR SELECT TO authenticated
USING (krijuar_nga = auth.uid() OR id IN (SELECT projekti_id FROM anetaret WHERE user_id = auth.uid()));

CREATE POLICY "detyrat_member_read" ON detyrat FOR SELECT TO authenticated
USING (projekti_id IN (SELECT id FROM projektet WHERE krijuar_nga = auth.uid())
OR projekti_id IN (SELECT projekti_id FROM anetaret WHERE user_id = auth.uid()));

CREATE POLICY "detyrat_member_update" ON detyrat FOR UPDATE TO authenticated
USING (projekti_id IN (SELECT id FROM projektet WHERE krijuar_nga = auth.uid())
OR projekti_id IN (SELECT projekti_id FROM anetaret WHERE user_id = auth.uid()));

CREATE POLICY "detyrat_member_insert" ON detyrat FOR INSERT TO authenticated
WITH CHECK (projekti_id IN (SELECT id FROM projektet WHERE krijuar_nga = auth.uid())
OR projekti_id IN (SELECT projekti_id FROM anetaret WHERE user_id = auth.uid()));

CREATE POLICY "anetaret_delete" ON anetaret FOR DELETE TO authenticated
USING (projekti_id IN (SELECT id FROM projektet WHERE krijuar_nga = auth.uid()));

CREATE POLICY "profiles_read_all" ON profiles FOR SELECT TO authenticated USING (true);

-- Trigger: auto-fill user_id when member invited by email
CREATE OR REPLACE FUNCTION fill_anetaret_user_id() RETURNS TRIGGER AS $$
BEGIN
  IF NEW.user_id IS NULL THEN
    SELECT id INTO NEW.user_id FROM auth.users WHERE LOWER(email) = LOWER(NEW.email) LIMIT 1;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
DROP TRIGGER IF EXISTS anetaret_fill_user_id ON anetaret;
CREATE TRIGGER anetaret_fill_user_id BEFORE INSERT ON anetaret FOR EACH ROW EXECUTE FUNCTION fill_anetaret_user_id();
ALTER TABLE anetaret ADD CONSTRAINT anetaret_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id);
